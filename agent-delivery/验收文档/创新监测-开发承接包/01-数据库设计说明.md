# 01 · 数据库设计说明（三仓契约面）

> **读者**：Python 仓（写入方）、Java 仓（只读方）
> **数据库**：MySQL 8.0，utf8mb4；图数据另存图库（见 05 文档），但图的快照同步到本库 `dwm_graph_*` 供排查与重建。
> **Schema 划分**：
> - `innov_dim`：维表/主数据（Python 写）
> - `innov_dwd` / `innov_dwm`：明细与中间层（Python 写）
> - `innov_ads`：应用层（Python 写，Java 只读，**契约面核心**）
> - `innov_cfg`：配置（运维/算法写）
> - `app_user`：用户态（Java 独立读写，订阅/推送/导出记录，本文仅给约定不给 DDL）
>
> **变更管理**：`innov_ads` 与 `innov_dim` 的结构变更需 Python/Java 双方会签；其余层 Python 自治。
> 命名规约：主键 `id` BIGINT 自增或业务键；时间统一 DATETIME；JSON 字段必须在注释中给出结构示例。

## 1. 维表 / 主数据（innov_dim）

```sql
-- 行政区划（含层级，支撑 GIS 多层级；边界 GeoJSON 大字段单独存）
CREATE TABLE dim_admin_division (
  admin_code      VARCHAR(20)  NOT NULL COMMENT '区划代码（国标12位，街道级补位）',
  name            VARCHAR(128) NOT NULL,
  level           TINYINT      NOT NULL COMMENT '1省 2市 3区县 4街道 5园区',
  parent_code     VARCHAR(20)  NULL,
  center_lng      DECIMAL(10,6) NULL,
  center_lat      DECIMAL(10,6) NULL,
  boundary_geo    LONGTEXT     NULL COMMENT 'GeoJSON 边界，园区/街道级按缺口G-4补充',
  PRIMARY KEY (admin_code), KEY idx_parent (parent_code)
) COMMENT='行政区划维表';

-- 全球城市维表（缺口G-24：海外城市先人工补录标杆城市包）
CREATE TABLE dim_city (
  city_code       VARCHAR(20)  NOT NULL COMMENT '自编码 CN-4403 / US-SFO 等',
  name_zh         VARCHAR(128) NOT NULL,
  name_en         VARCHAR(128) NULL,
  country_code    VARCHAR(8)   NOT NULL,
  admin_code      VARCHAR(20)  NULL COMMENT '国内城市关联区划',
  lng             DECIMAL(10,6) NULL,
  lat             DECIMAL(10,6) NULL,
  is_benchmark    TINYINT      NOT NULL DEFAULT 0 COMMENT '是否一期标杆城市',
  PRIMARY KEY (city_code)
) COMMENT='全球城市维表';

-- 技术/领域树（自建领域层：产业链节点 + IPC映射 + 学科映射，缺口G-10/G-11）
CREATE TABLE dim_tech_domain (
  domain_id       VARCHAR(64)  NOT NULL,
  name            VARCHAR(255) NOT NULL,
  parent_id       VARCHAR(64)  NULL,
  level           TINYINT      NOT NULL,
  domain_source   VARCHAR(32)  NOT NULL COMMENT 'chain=产业链表 ipc=IPC映射 subject=学科 kgp=亿级图谱平台(预留)',
  source_code     VARCHAR(64)  NULL COMMENT '来源体系内编码（chain_code/node_id、IPC号、学科代码）',
  PRIMARY KEY (domain_id), KEY idx_parent (parent_id)
) COMMENT='领域/技术树（展示与统计的统一领域口径）';

-- 事件类型字典
CREATE TABLE dim_event_type (
  event_type      VARCHAR(32)  NOT NULL COMMENT 'tech_break/achievement/financing/invest/ma/bid/biz_dynamic/policy/fund_project/talent_flow/award/industry_event',
  name_zh         VARCHAR(64)  NOT NULL,
  color           VARCHAR(16)  NOT NULL COMMENT '前端地图图例色',
  produce_mode    VARCHAR(16)  NOT NULL COMMENT 'rule=结构化直转 nlp=资讯加工',
  importance_prior DECIMAL(4,3) NOT NULL DEFAULT 0.5 COMMENT '类型先验分，参与重要性计算',
  PRIMARY KEY (event_type)
) COMMENT='事件类型字典';

-- 机构主数据（全局ID贯通产物）
CREATE TABLE dim_org_master (
  org_uid         VARCHAR(64)  NOT NULL COMMENT '统一机构ID（本系统主键）',
  name_std        VARCHAR(512) NOT NULL COMMENT '标准名称',
  credit_code     VARCHAR(32)  NULL,
  org_type        VARCHAR(32)  NOT NULL COMMENT 'company/university/research/carrier/overseas/gov',
  country_code    VARCHAR(8)   NOT NULL DEFAULT 'CN',
  city_code       VARCHAR(20)  NULL,
  admin_code      VARCHAR(20)  NULL,
  lng             DECIMAL(10,6) NULL,
  lat             DECIMAL(10,6) NULL,
  parent_org_uid  VARCHAR(64)  NULL COMMENT '母公司',
  src_org_id      VARCHAR(64)  NULL COMMENT '来源: dwd_org_reg_info.org_id',
  src_company_id  VARCHAR(64)  NULL COMMENT '来源: 量子专题 dwd_company_base.id',
  src_forg_id     VARCHAR(64)  NULL COMMENT '来源: dwd_forg_base_info.org_id',
  updated_time    DATETIME     NOT NULL,
  PRIMARY KEY (org_uid), KEY idx_credit (credit_code), KEY idx_city (city_code)
) COMMENT='机构主数据（ID贯通）';

CREATE TABLE dim_org_alias (
  id              BIGINT AUTO_INCREMENT,
  org_uid         VARCHAR(64)  NOT NULL,
  alias           VARCHAR(512) NOT NULL,
  alias_type      VARCHAR(16)  NOT NULL COMMENT 'history=曾用名 en=英文 abbr=简称 paper=论文署名变体 patent=专利署名变体',
  confidence      DECIMAL(4,3) NOT NULL DEFAULT 1.0,
  PRIMARY KEY (id), KEY idx_org (org_uid), KEY idx_alias (alias(191))
) COMMENT='机构别名表';

-- 创新载体名录（一期=高校+资质标签企业近似；补数后扩展，缺口G-13/14）
CREATE TABLE dim_carrier (
  carrier_id      VARCHAR(64)  NOT NULL,
  name            VARCHAR(512) NOT NULL,
  carrier_type    VARCHAR(32)  NOT NULL COMMENT 'university/research/incubator/maker_space/key_lab/eng_center/park',
  carrier_level   VARCHAR(16)  NULL COMMENT 'national/provincial/city',
  operator_org_uid VARCHAR(64) NULL COMMENT '运营主体',
  admin_code      VARCHAR(20)  NULL,
  lng             DECIMAL(10,6) NULL,
  lat             DECIMAL(10,6) NULL,
  source          VARCHAR(32)  NOT NULL COMMENT 'v1_approx=资质标签近似 supplier=供方名录',
  PRIMARY KEY (carrier_id)
) COMMENT='创新载体名录';

CREATE TABLE dim_carrier_org_rel (
  id              BIGINT AUTO_INCREMENT,
  carrier_id      VARCHAR(64)  NOT NULL,
  org_uid         VARCHAR(64)  NOT NULL,
  rel_type        VARCHAR(16)  NOT NULL COMMENT 'self=载体即主体 tenant=入驻 graduated=毕业',
  start_date      DATE NULL, end_date DATE NULL,
  attribution_coef DECIMAL(4,3) NOT NULL DEFAULT 1.0 COMMENT '成果归属系数（入驻企业默认0.5，配置可调）',
  PRIMARY KEY (id), KEY idx_carrier (carrier_id), KEY idx_org (org_uid)
) COMMENT='载体-主体关系（缺口G-14到位前 rel_type 仅 self）';
```

## 2. 明细层（innov_dwd）

```sql
-- 标准事件明细（结构化直转 + NLP 加工统一入此表，是 ads 事件表的唯一上游）
CREATE TABLE dwd_event_std (
  event_id        VARCHAR(64)  NOT NULL COMMENT 'md5(来源表+来源主键) 或聚合后新ID',
  event_type      VARCHAR(32)  NOT NULL,
  sub_type        VARCHAR(64)  NULL,
  title           VARCHAR(1024) NOT NULL,
  summary         TEXT NULL,
  content         LONGTEXT NULL,
  event_time      DATETIME NULL COMMENT '业务时间；缺失时 time_quality 标记',
  time_quality    VARCHAR(16) NOT NULL DEFAULT 'exact' COMMENT 'exact/approx_updated_time/extracted',
  country_code    VARCHAR(8) NULL,
  city_code       VARCHAR(20) NULL,
  admin_code      VARCHAR(20) NULL,
  lng             DECIMAL(10,6) NULL,
  lat             DECIMAL(10,6) NULL,
  geo_source      VARCHAR(16) NULL COMMENT 'org_reg=主体注册地 extracted=正文抽取',
  amount          DECIMAL(20,2) NULL COMMENT '金额类特征（融资/并购/中标）',
  currency        VARCHAR(8) NULL,
  orgs            JSON NULL COMMENT '[{"org_uid":"","name":"","role":"main/investor/acquired/winner"}]',
  persons         JSON NULL COMMENT '[{"name":"","scholar_id":null,"role":""}]',
  domains         JSON NULL COMMENT '["domain_id",...]',
  source_refs     JSON NOT NULL COMMENT '[{"ods_table":"","pk":"","url":""}] 多来源聚合',
  source_count    INT NOT NULL DEFAULT 1 COMMENT '来源数（热度降级特征）',
  dedup_key       VARCHAR(64) NOT NULL COMMENT 'SimHash/规则键',
  produce_mode    VARCHAR(16) NOT NULL COMMENT 'rule/nlp',
  model_version   VARCHAR(32) NULL,
  confidence      DECIMAL(4,3) NULL,
  created_time    DATETIME NOT NULL, updated_time DATETIME NOT NULL,
  PRIMARY KEY (event_id),
  KEY idx_type_time (event_type, event_time), KEY idx_dedup (dedup_key), KEY idx_city (city_code)
) COMMENT='标准事件明细';
```

## 3. 中间层（innov_dwm）

```sql
-- 图谱节点/边（图库为运行时主存，此处为可重建快照）
CREATE TABLE dwm_graph_node (
  node_id    VARCHAR(64) NOT NULL,
  node_type  VARCHAR(16) NOT NULL COMMENT 'org/scholar/paper/patent/project/domain/event',
  ref_id     VARCHAR(64) NOT NULL COMMENT '指向各主数据/明细主键',
  name       VARCHAR(512) NULL,
  props      JSON NULL,
  PRIMARY KEY (node_id), KEY idx_type_ref (node_type, ref_id)
) COMMENT='图谱节点快照';

CREATE TABLE dwm_graph_edge (
  edge_id    BIGINT AUTO_INCREMENT,
  src_id     VARCHAR(64) NOT NULL,
  dst_id     VARCHAR(64) NOT NULL,
  edge_type  VARCHAR(32) NOT NULL COMMENT 'authored/cites/assignee/funded/coop/invest/chain_belong/event_about',
  weight     DECIMAL(8,4) NOT NULL DEFAULT 1,
  event_date DATE NULL COMMENT '用于时间衰减',
  props      JSON NULL,
  PRIMARY KEY (edge_id), KEY idx_src (src_id, edge_type), KEY idx_dst (dst_id, edge_type)
) COMMENT='图谱边快照';

-- 主体成果年度宽表（指数计算的统一输入；从论文/专利/项目明细按年聚合，绕开 G-18 快照表无年度问题）
CREATE TABLE dwm_org_achievement_year (
  org_uid         VARCHAR(64) NOT NULL,
  stat_year       SMALLINT    NOT NULL,
  domain_id       VARCHAR(64) NOT NULL DEFAULT 'ALL',
  paper_cnt INT DEFAULT 0, paper_sci_cnt INT DEFAULT 0,
  patent_apply_cnt INT DEFAULT 0, patent_grant_inv_cnt INT DEFAULT 0, patent_pct_cnt INT DEFAULT 0,
  project_cnt INT DEFAULT 0, project_amount DECIMAL(20,2) DEFAULT 0,
  financing_cnt INT DEFAULT 0, financing_amount DECIMAL(20,2) DEFAULT 0,
  event_cnt INT DEFAULT 0,
  updated_time DATETIME NOT NULL,
  PRIMARY KEY (org_uid, stat_year, domain_id)
) COMMENT='主体-成果年度宽表';
```

## 4. 应用层（innov_ads，Java 只读）

```sql
-- M1 科创事件（前端地图/列表/详情直接消费）
CREATE TABLE ads_innovation_event (
  event_id        VARCHAR(64) NOT NULL,
  event_type      VARCHAR(32) NOT NULL,
  title           VARCHAR(1024) NOT NULL,
  summary         TEXT NULL,
  event_time      DATETIME NOT NULL,
  importance_level TINYINT NOT NULL COMMENT '1高 2中 3低',
  importance_score DECIMAL(6,2) NOT NULL,
  importance_detail JSON NULL COMMENT '{"主体影响分":x,"热度":x,...} 可解释明细',
  heat_score      DECIMAL(6,2) NOT NULL DEFAULT 0,
  country_code VARCHAR(8) NULL, city_code VARCHAR(20) NULL, admin_code VARCHAR(20) NULL,
  lng DECIMAL(10,6) NULL, lat DECIMAL(10,6) NULL,
  orgs JSON NULL, persons JSON NULL, domains JSON NULL,
  source_refs JSON NOT NULL,
  model_version VARCHAR(32) NULL,
  updated_time DATETIME NOT NULL,
  PRIMARY KEY (event_id),
  KEY idx_time_level (event_time, importance_level),
  KEY idx_type_time (event_type, event_time),
  KEY idx_geo (city_code, event_time)
) COMMENT='科创事件应用表';

CREATE TABLE ads_event_relation (
  id BIGINT AUTO_INCREMENT,
  event_id VARCHAR(64) NOT NULL,
  rel_type VARCHAR(16) NOT NULL COMMENT 'event/org/person',
  rel_id   VARCHAR(64) NOT NULL,
  rel_name VARCHAR(512) NULL,
  rel_desc VARCHAR(255) NULL COMMENT '同主体/同主题/共现…',
  score    DECIMAL(4,3) NULL,
  PRIMARY KEY (id), KEY idx_event (event_id)
) COMMENT='事件关联（详情页关联分析）';

-- M2 节点影响分
CREATE TABLE ads_node_influence (
  node_id VARCHAR(64) NOT NULL,
  node_type VARCHAR(16) NOT NULL,
  name VARCHAR(512) NOT NULL,
  domain_id VARCHAR(64) NOT NULL,
  stat_period VARCHAR(7) NOT NULL COMMENT '快照期 yyyy-MM',
  influence_score DECIMAL(8,2) NOT NULL,
  rank_in_domain INT NOT NULL,
  score_detail JSON NOT NULL COMMENT '{"ppr":x,"degree":x,"betweenness":x,"output":x} 分项贡献',
  PRIMARY KEY (node_id, domain_id, stat_period),
  KEY idx_rank (domain_id, stat_period, rank_in_domain)
) COMMENT='实体节点创新影响分';

-- 创新点落图结果
CREATE TABLE ads_innovation_point_map (
  id BIGINT AUTO_INCREMENT,
  point_id VARCHAR(64) NOT NULL COMMENT '创新点=事件/论文/专利/项目',
  point_type VARCHAR(16) NOT NULL,
  node_id VARCHAR(64) NOT NULL COMMENT '映射到的图谱节点',
  rel_type VARCHAR(32) NOT NULL,
  confidence DECIMAL(4,3) NOT NULL,
  model_version VARCHAR(32) NULL,
  updated_time DATETIME NOT NULL,
  PRIMARY KEY (id), KEY idx_point (point_id), KEY idx_node (node_id)
) COMMENT='创新点-图谱节点映射';

-- M3 载体指数（按期快照，支撑趋势曲线）
CREATE TABLE ads_carrier_index (
  carrier_id VARCHAR(64) NOT NULL,
  stat_period VARCHAR(7) NOT NULL,
  domain_id VARCHAR(64) NOT NULL DEFAULT 'ALL',
  index_score DECIMAL(8,2) NOT NULL,
  rank_total INT NULL, rank_domain INT NULL, rank_region INT NULL,
  sub_scores JSON NOT NULL COMMENT '{"output":x,"quality":x,"activity":x,"performance":x}',
  weight_version VARCHAR(32) NOT NULL COMMENT '关联 cfg_index_weight，需规要求权重可查',
  PRIMARY KEY (carrier_id, stat_period, domain_id),
  KEY idx_rank (stat_period, domain_id, rank_total)
) COMMENT='创新载体能力指数';

-- M4 城市指数
CREATE TABLE ads_city_index (
  city_code VARCHAR(20) NOT NULL,
  stat_period VARCHAR(7) NOT NULL,
  index_score DECIMAL(8,2) NOT NULL,
  rank_global INT NULL, rank_domestic INT NULL,
  sub_scores JSON NOT NULL COMMENT '{"resource":x,"institution":x,"output":x,"environment":x}',
  indicator_values JSON NOT NULL COMMENT '指标级数值，对标页明细用',
  weight_version VARCHAR(32) NOT NULL,
  PRIMARY KEY (city_code, stat_period)
) COMMENT='城市创新能力指数';

-- M5 预测结果
CREATE TABLE ads_impact_prediction (
  id BIGINT AUTO_INCREMENT,
  target_type VARCHAR(16) NOT NULL COMMENT 'domain=领域趋势 point=创新点贡献度',
  target_id VARCHAR(64) NOT NULL,
  domain_id VARCHAR(64) NOT NULL,
  predict_period VARCHAR(7) NOT NULL,
  predict_value DECIMAL(10,4) NOT NULL,
  ci_lower DECIMAL(10,4) NULL, ci_upper DECIMAL(10,4) NULL COMMENT '置信区间（需规硬性要求）',
  contribution DECIMAL(6,4) NULL COMMENT '贡献度（point 类）',
  explain_path JSON NULL COMMENT '可解释传播路径',
  model_version VARCHAR(32) NOT NULL,
  created_time DATETIME NOT NULL,
  PRIMARY KEY (id), KEY idx_target (target_type, target_id, predict_period)
) COMMENT='创新影响预测';

CREATE TABLE ads_topic_burst (
  id BIGINT AUTO_INCREMENT,
  domain_id VARCHAR(64) NOT NULL,
  topic_name VARCHAR(255) NOT NULL,
  keywords JSON NOT NULL,
  burst_start DATE NOT NULL, burst_score DECIMAL(8,2) NOT NULL,
  evidence JSON NULL COMMENT '支撑论文/专利ID列表',
  stat_period VARCHAR(7) NOT NULL,
  PRIMARY KEY (id), KEY idx_domain (domain_id, stat_period)
) COMMENT='主题突变/颠覆性技术检测';

-- 报告类（政策建议/竞争力分析，生成后落库，Java 直接取）
CREATE TABLE ads_report (
  report_id VARCHAR(64) NOT NULL,
  report_type VARCHAR(32) NOT NULL COMMENT 'city_policy_advice/city_benchmark/value_assess',
  target_id VARCHAR(64) NOT NULL,
  title VARCHAR(512) NOT NULL,
  content_md LONGTEXT NOT NULL COMMENT 'Markdown 正文',
  generate_mode VARCHAR(16) NOT NULL COMMENT 'template/llm',
  review_status TINYINT NOT NULL DEFAULT 0 COMMENT '0待审 1已发布',
  created_time DATETIME NOT NULL,
  PRIMARY KEY (report_id), KEY idx_target (report_type, target_id)
) COMMENT='生成式报告';
```

## 5. 配置层（innov_cfg）

```sql
-- ★ 数据缺口降级开关（缺口清单 G-x 的工程化）
CREATE TABLE cfg_degrade_switch (
  switch_key VARCHAR(64) NOT NULL COMMENT '如 heat_source / ma_event_time / carrier_scope / city_scope',
  gap_ref    VARCHAR(16) NOT NULL COMMENT '关联缺口编号 G-1..G-25',
  cur_value  VARCHAR(64) NOT NULL COMMENT '当前策略，如 heat_source=source_count|real_metrics',
  options    JSON NOT NULL,
  remark     VARCHAR(512) NULL,
  updated_time DATETIME NOT NULL,
  PRIMARY KEY (switch_key)
) COMMENT='缺口降级开关：补数后改值，不改代码';

-- 指数权重（需规要求"用户可查看计算方法和权重"→ Java 有只读接口）
CREATE TABLE cfg_index_weight (
  weight_version VARCHAR(32) NOT NULL,
  index_type VARCHAR(16) NOT NULL COMMENT 'carrier/city',
  weights JSON NOT NULL COMMENT '层级权重树 + 指标说明文案',
  method_desc TEXT NOT NULL COMMENT '计算方法说明（前端弹窗展示）',
  effective_date DATE NOT NULL,
  PRIMARY KEY (weight_version, index_type)
) COMMENT='指数权重与方法说明';

CREATE TABLE cfg_model_registry (
  model_version VARCHAR(32) NOT NULL,
  task VARCHAR(32) NOT NULL COMMENT 'event_cls/event_extract/entity_link/importance/forecast',
  engine VARCHAR(16) NOT NULL COMMENT 'llm/bert/rule',
  metrics JSON NULL COMMENT '评测结果',
  is_active TINYINT NOT NULL DEFAULT 0,
  created_time DATETIME NOT NULL,
  PRIMARY KEY (model_version, task)
) COMMENT='模型版本登记（双轨切换记录）';
```

## 6. 用户态（app_user，Java 自治，约定）

- `user_subscription`（订阅：user_id、sub_type[domain/carrier/city/keyword]、sub_target、频率、渠道）；
- `push_record`（推送历史）；`export_record`（导出/分享）。
- Java 定时任务扫描 `innov_ads` 增量（按 updated_time）匹配订阅规则触发推送；**不回写 innov 库**。

## 7. 数据流转与刷新频率约定

| 表 | 写入方 | 频率 |
|---|---|---|
| dim_*（机构/别名/载体） | Python S1 流水线 | 日增量 |
| dwd_event_std → ads_innovation_event | Python 事件流水线 | 小时级（资讯入库→可见 ≤1h） |
| ads_node_influence / ads_carrier_index / ads_city_index | Python 批计算 | 月度快照（period=yyyy-MM），事件热度日内重算 |
| ads_impact_prediction / ads_topic_burst | Python | 月度 |
| ads_report | Python（生成）→ 人工审核置 review_status | 月度/按需 |
