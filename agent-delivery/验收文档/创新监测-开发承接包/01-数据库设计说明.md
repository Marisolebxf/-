# 01 · 数据库设计说明（三仓契约面）

> **读者**：Python 仓（写入方）、Java 仓（只读方）  
> **数据库**：PostgreSQL 15+，库 `demo_gdf_data`（utf8）；图数据另存 NebulaGraph（见 05 文档），图快照同步到本库 `dwd.graph_*` 供排查与重建。  
> **Schema 划分（1B+2B：同库多 schema，去层前缀）**：
> - `dim`：维表/主数据（Python 写）← 原 `innov_dim` / `dim_*`
> - `dwd`：明细与中间层（Python 写）← 原 `innov_dwd` / `innov_dwm` / `dwd_*` / `dwm_*`
> - `ads`：应用层（Python 写，Java 只读，**契约面核心**）← 原 `innov_ads` / `ads_*`
> - `gkx_cfg`：配置（运维/算法写）← 原 `innov_cfg` / `cfg_*`
> - `ods`：三方贴源镜像（按需）
> - `app_user`：用户态（Java 独立读写，订阅/推送/导出记录，本文仅给约定不给 DDL）
>
> **变更管理**：`ads` 与 `dim` 的结构变更需 Python/Java 双方会签；其余层 Python 自治。  
> **命名规约**：主键 `id` 用 `bigserial` 或业务键；时间统一 `timestamp`/`timestamptz`；JSON 用 `jsonb`，结构示例写在注释/设计说明中。  
> **真源 DDL**：`demo-algo-science/scripts/sql/pg/schema_pg.sql` + `innov_table_map.yaml`。

## 1. 维表 / 主数据（`dim`）

```sql
CREATE SCHEMA IF NOT EXISTS dim;

-- 行政区划（含层级，支撑 GIS 多层级；边界 GeoJSON 大字段单独存）
CREATE TABLE IF NOT EXISTS dim.admin_division (
  admin_code   varchar(20)  NOT NULL,  -- 区划代码（国标12位，街道级补位）
  name         varchar(128) NOT NULL,
  level        smallint     NOT NULL,  -- 1省 2市 3区县 4街道 5园区
  parent_code  varchar(20)  NULL,
  center_lng   numeric(10,6) NULL,
  center_lat   numeric(10,6) NULL,
  boundary_geo text         NULL,      -- GeoJSON 边界，园区/街道级按缺口 G-4 补充
  PRIMARY KEY (admin_code)
);
CREATE INDEX IF NOT EXISTS idx_dim_admin_parent ON dim.admin_division (parent_code);
COMMENT ON TABLE dim.admin_division IS '行政区划维表';

-- 全球城市维表（缺口 G-24：海外城市先人工补录标杆城市包）
CREATE TABLE IF NOT EXISTS dim.city (
  city_code    varchar(20)  NOT NULL,  -- 自编码 CN-4403 / US-SFO 等
  name_zh      varchar(128) NOT NULL,
  name_en      varchar(128) NULL,
  country_code varchar(8)   NOT NULL,
  admin_code   varchar(20)  NULL,      -- 国内城市关联区划
  lng          numeric(10,6) NULL,
  lat          numeric(10,6) NULL,
  is_benchmark smallint     NOT NULL DEFAULT 0,  -- 是否一期标杆城市
  PRIMARY KEY (city_code)
);
COMMENT ON TABLE dim.city IS '全球城市维表';

-- 技术/领域树（产业链节点 + IPC 映射 + 学科映射，缺口 G-10/G-11）
CREATE TABLE IF NOT EXISTS dim.tech_domain (
  domain_id     varchar(64)  NOT NULL,
  name          varchar(255) NOT NULL,
  parent_id     varchar(64)  NULL,
  level         smallint     NOT NULL,
  domain_source varchar(32)  NOT NULL,  -- chain / ipc / subject / kgp
  source_code   varchar(64)  NULL,
  PRIMARY KEY (domain_id)
);
CREATE INDEX IF NOT EXISTS idx_dim_tech_parent ON dim.tech_domain (parent_id);
COMMENT ON TABLE dim.tech_domain IS '领域/技术树（展示与统计的统一领域口径）';

-- 事件类型字典
CREATE TABLE IF NOT EXISTS dim.event_type (
  event_type        varchar(32)  NOT NULL,  -- tech_break/achievement/financing/...
  name_zh           varchar(64)  NOT NULL,
  color             varchar(16)  NOT NULL,  -- 前端地图图例色
  produce_mode      varchar(16)  NOT NULL,  -- rule / nlp
  importance_prior  numeric(4,3) NOT NULL DEFAULT 0.5,
  PRIMARY KEY (event_type)
);
COMMENT ON TABLE dim.event_type IS '事件类型字典';

-- 机构主数据（全局 ID 贯通）
CREATE TABLE IF NOT EXISTS dim.org_master (
  org_uid        varchar(64)  NOT NULL,
  name_std       varchar(512) NOT NULL,
  credit_code    varchar(32)  NULL,
  org_type       varchar(32)  NOT NULL,  -- company/university/research/carrier/overseas/gov
  country_code   varchar(8)   NOT NULL DEFAULT 'CN',
  city_code      varchar(20)  NULL,
  admin_code     varchar(20)  NULL,
  lng            numeric(10,6) NULL,
  lat            numeric(10,6) NULL,
  parent_org_uid varchar(64)  NULL,
  src_org_id     varchar(64)  NULL,      -- ODS: dwd_org_reg_info.org_id（供应方表名，非本库 schema）
  src_company_id varchar(64)  NULL,
  src_forg_id    varchar(64)  NULL,
  updated_time   timestamp    NOT NULL,
  PRIMARY KEY (org_uid)
);
CREATE INDEX IF NOT EXISTS idx_dim_org_credit ON dim.org_master (credit_code);
CREATE INDEX IF NOT EXISTS idx_dim_org_city ON dim.org_master (city_code);
COMMENT ON TABLE dim.org_master IS '机构主数据（ID贯通）';

CREATE TABLE IF NOT EXISTS dim.org_alias (
  id          bigserial PRIMARY KEY,
  org_uid     varchar(64)  NOT NULL,
  alias       varchar(512) NOT NULL,
  alias_type  varchar(16)  NOT NULL,  -- history/en/abbr/paper/patent
  confidence  numeric(4,3) NOT NULL DEFAULT 1.0
);
CREATE INDEX IF NOT EXISTS idx_dim_alias_org ON dim.org_alias (org_uid);
CREATE INDEX IF NOT EXISTS idx_dim_alias_text ON dim.org_alias (alias);
COMMENT ON TABLE dim.org_alias IS '机构别名表';

-- 创新载体名录（一期=高校+资质标签企业近似；补数后扩展，缺口 G-13/14）
CREATE TABLE IF NOT EXISTS dim.carrier (
  carrier_id       varchar(64)  NOT NULL,
  name             varchar(512) NOT NULL,
  carrier_type     varchar(32)  NOT NULL,
  carrier_level    varchar(16)  NULL,   -- national/provincial/city
  operator_org_uid varchar(64)  NULL,
  admin_code       varchar(20)  NULL,
  lng              numeric(10,6) NULL,
  lat              numeric(10,6) NULL,
  source           varchar(32)  NOT NULL,  -- v1_approx / supplier
  PRIMARY KEY (carrier_id)
);
COMMENT ON TABLE dim.carrier IS '创新载体名录';

CREATE TABLE IF NOT EXISTS dim.carrier_org_rel (
  id               bigserial PRIMARY KEY,
  carrier_id       varchar(64)  NOT NULL,
  org_uid          varchar(64)  NOT NULL,
  rel_type         varchar(16)  NOT NULL,  -- self/tenant/graduated
  start_date       date NULL,
  end_date         date NULL,
  attribution_coef numeric(4,3) NOT NULL DEFAULT 1.0
);
CREATE INDEX IF NOT EXISTS idx_dim_carrier_rel_c ON dim.carrier_org_rel (carrier_id);
CREATE INDEX IF NOT EXISTS idx_dim_carrier_rel_o ON dim.carrier_org_rel (org_uid);
COMMENT ON TABLE dim.carrier_org_rel IS '载体-主体关系';
```

## 2. 明细层（`dwd`）

```sql
CREATE SCHEMA IF NOT EXISTS dwd;

-- 标准事件明细（结构化直转 + NLP 加工统一入此表，是 ads 事件表的唯一上游）
CREATE TABLE IF NOT EXISTS dwd.event_std (
  event_id      varchar(64)   NOT NULL,
  event_type    varchar(32)   NOT NULL,
  sub_type      varchar(64)   NULL,
  title         varchar(1024) NOT NULL,
  summary       text NULL,
  content       text NULL,
  event_time    timestamp NULL,
  time_quality  varchar(32)   NOT NULL DEFAULT 'exact',  -- exact/approx_updated_time/extracted
  country_code  varchar(8) NULL,
  city_code     varchar(20) NULL,
  admin_code    varchar(20) NULL,
  lng           numeric(10,6) NULL,
  lat           numeric(10,6) NULL,
  geo_source    varchar(16) NULL,      -- org_reg / extracted

  orgs          jsonb NULL,           -- [{"org_uid","name","role"}]

  dedup_key     varchar(64) NOT NULL,
  produce_mode  varchar(16) NOT NULL,  -- rule/nlp

CREATE INDEX IF NOT EXISTS idx_dwd_event_type_time ON dwd.event_std (event_type, event_time);
```

```sql
COMMENT ON TABLE ads.innovation_point_map IS '创新点-图谱节点映射';
```

## 6. 用户态（`app_user`，Java 自治，约定）

- `user_subscription`（订阅：user_id、sub_type[domain/carrier/city/keyword]、sub_target、频率、渠道）；
- `push_record`（推送历史）；`export_record`（导出/分享）。
- Java 定时任务扫描 `ads.*` 增量（按 `updated_time`）匹配订阅规则触发推送；**不回写 innov 业务 schema**。

## 7. 数据流转与刷新频率约定

| 表 | 写入方 | 频率 |
|---|---|---|
| `dim.org_*` / `dim.carrier*` | Python S1 流水线 | 日增量 |
| `dwd.event_std` → `ads.innovation_event` | Python 事件流水线 | 小时级（资讯入库→可见 ≤1h） |
| `ads.node_influence` / `ads.carrier_index` / `ads.city_index` | Python 批计算 | 月度快照（period=yyyy-MM），事件热度日内重算 |
| `ads.impact_prediction` / `ads.topic_burst` | Python | 月度 |
| `ads.report` | Python（生成）→ 人工审核置 `review_status` | 月度/按需 |
